<script>
  let pieces = [];
  let activePiece = null;

  // Comprime imagem para max 1200px e retorna { base64, mimeType }
  function compressImage(file) {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onerror = reject;
      reader.onload = e => {
        const img = new Image();
        img.onerror = reject;
        img.onload = () => {
          const MAX = 1200;
          let { width, height } = img;
          if (width > MAX || height > MAX) {
            if (width > height) { height = Math.round(height * MAX / width); width = MAX; }
            else                { width  = Math.round(width  * MAX / height); height = MAX; }
          }
          const canvas = document.createElement('canvas');
          canvas.width = width; canvas.height = height;
          canvas.getContext('2d').drawImage(img, 0, 0, width, height);
          const dataUrl = canvas.toDataURL('image/jpeg', 0.82);
          const base64  = dataUrl.split(',')[1];
          resolve({ base64, mimeType: 'image/jpeg' });
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    });
  }

  function showToast(msg, duration = 2500) {
    const t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(() => t.classList.remove('show'), duration);
  }

  function updateCounter() {
    const remaining = pieces.filter(p => !p.done).length;
    document.getElementById('counter').textContent =
      remaining > 0 ? `${remaining} sem foto` : 'tudo com foto ✓';
  }

  function renderList() {
    const app = document.getElementById('app');
    const remaining = pieces.filter(p => !p.done);

    if (!remaining.length) {
      app.innerHTML = `<div class="empty">🎉 Todas as peças têm foto!</div>`;
      updateCounter();
      return;
    }

    app.innerHTML = `<div class="list">${remaining.map(p => `
      <div class="piece-card ${p.done ? 'done' : ''}" data-row="${p.row}" id="card-${p.row}">
        <div class="thumb" id="thumb-${p.row}">
          ${p.localPreview
            ? `<img src="${p.localPreview}">`
            : '👗'}
        </div>
        <div class="piece-info">
          <div class="piece-id">${p.id}</div>
          <div class="piece-desc">${p.desc}</div>
          <div class="piece-meta">${[p.brand, p.size, formatPrice(p.price)].filter(Boolean).join(' · ')}</div>
        </div>
        <div class="add-icon" id="icon-${p.row}">
          ${p.done ? '✓' : '+'}
        </div>
      </div>`).join('')}
    </div>`;

    updateCounter();

    document.querySelectorAll('.piece-card').forEach(card => {
      card.addEventListener('click', () => {
        const row = parseInt(card.dataset.row);
        activePiece = pieces.find(p => p.row === row);
        document.getElementById('filePicker').value = '';
        document.getElementById('filePicker').click();
      });
    });
  }

  function formatPrice(raw) {
    const n = parseFloat(String(raw).replace(/[R$\s"]/g, '').replace(',', '.'));
    if (isNaN(n)) return '';
    return 'R$ ' + n.toLocaleString('pt-BR', { minimumFractionDigits: 0 });
  }

  document.getElementById('filePicker').addEventListener('change', async e => {
    const file = e.target.files[0];
    if (!file || !activePiece) return;

    const piece = activePiece;
    const card  = document.getElementById('card-' + piece.row);
    const icon  = document.getElementById('icon-' + piece.row);
    const thumb = document.getElementById('thumb-' + piece.row);

    // Preview local imediato
    const localUrl = URL.createObjectURL(file);
    thumb.innerHTML = `<img src="${localUrl}">`;
    piece.localPreview = localUrl;

    // Spinner no lugar do ícone
    card.classList.add('uploading');
    icon.innerHTML = '<div class="spinner"></div>';

    try {
      const { base64, mimeType } = await compressImage(file);

      const url = await new Promise((resolve, reject) => {
        google.script.run
          .withSuccessHandler(resolve)
          .withFailureHandler(reject)
          .savePhoto(piece.row, base64, mimeType, piece.id);
      });

      piece.done = true;
      card.classList.remove('uploading');
      card.classList.add('done');
      icon.textContent = '✓';
      showToast(`${piece.id} salvo ✓`);
      updateCounter();

      // Remove o card após 1.5s
      setTimeout(() => {
        card.style.transition = 'opacity 0.3s';
        card.style.opacity = '0';
        setTimeout(() => card.remove(), 300);
        const remaining = pieces.filter(p => !p.done);
        if (!remaining.length) {
          document.getElementById('app').innerHTML =
            `<div class="empty">🎉 Todas as peças têm foto!</div>`;
        }
      }, 1500);

    } catch (err) {
      card.classList.remove('uploading');
      icon.textContent = '+';
      thumb.innerHTML = '👗';
      piece.localPreview = null;
      showToast('Erro ao salvar. Tente novamente.');
      console.error(err);
    }
  });

  // Carrega peças
  google.script.run
    .withSuccessHandler(data => {
      pieces = data;
      renderList();
    })
    .withFailureHandler(err => {
      document.getElementById('app').innerHTML =
        `<div class="empty">Erro ao carregar peças.<br><small>${err.message}</small></div>`;
    })
    .getPieces();
</script>
